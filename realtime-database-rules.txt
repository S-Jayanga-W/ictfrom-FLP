// =========================================================
// REALTIME DATABASE SECURITY RULES
// -----------------------------------------------------------
// HOW TO APPLY:
//   1. Go to Firebase Console -> Realtime Database -> Rules tab
//      (NOT "Firestore Database" — this project uses the older
//       Realtime Database, since that's what index log.html uses)
//   2. Delete everything there and paste this in
//   3. Replace the two "REPLACE_WITH_ADMIN_UID" below with your
//      real admin UID: oNftxzL7oJMYBf4bzEZgBMBZqKx2
//   4. Click "Publish"
// =========================================================
//
// WHY THIS MATTERS:
// Your existing "users/{uid}" data (name, phone, school, etc.)
// is completely UNTOUCHED by this — index log.html keeps working
// exactly as it does now.
//
// The lock/unlock state lives in a brand-new separate place:
// "unlocked/{uid}/{lessonId}" = true.
//
// These rules make sure:
//   - A student can read their OWN unlocked-lessons list only
//     (so lessons.html can check what they've paid for)
//   - A student can NEVER write to "unlocked" themselves — only
//     the admin account can, which is exactly what admin.html does
//   - The admin can read/write everything (the whole student list,
//     and everyone's unlock status)

{
  "rules": {

    "users": {
      ".read": "auth != null && auth.uid === 'oNftxzL7oJMYBf4bzEZgBMBZqKx2'",
      "$uid": {
        ".read": "auth != null && auth.uid === $uid",
        ".write": "auth != null && ((auth.uid === $uid && !data.exists()) || auth.uid === 'oNftxzL7oJMYBf4bzEZgBMBZqKx2')"
      }
    },

    "unlocked": {
      ".read": "auth != null && auth.uid === 'oNftxzL7oJMYBf4bzEZgBMBZqKx2'",
      "$uid": {
        ".read": "auth != null && auth.uid === $uid",
        ".write": "auth != null && auth.uid === 'oNftxzL7oJMYBf4bzEZgBMBZqKx2'"
      }
    }

  }
}
